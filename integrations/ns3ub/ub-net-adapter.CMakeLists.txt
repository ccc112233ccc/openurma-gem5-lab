set(ubsim_lab_root "${UBSIM_LAB_ROOT}")
if(NOT ubsim_lab_root)
  message(FATAL_ERROR "UBSIM_LAB_ROOT must name the ubsim-gem5-lab checkout")
endif()

add_library(ubsim_simbricks_base STATIC simbricks-base-portable.c)
target_include_directories(ubsim_simbricks_base SYSTEM PUBLIC
  "${ubsim_lab_root}/sources/simbricks/lib")

build_exec(
  EXECNAME ub-net-adapter
  EXECNAME_PREFIX scratch_
  SOURCE_FILES ${CMAKE_CURRENT_SOURCE_DIR}/ub-net-adapter.cc
  LIBRARIES_TO_LINK ${libunified-bus} ${libcore} ${libnetwork}
                    ubsim_simbricks_base
  EXECUTABLE_DIRECTORY_PATH ${CMAKE_OUTPUT_DIRECTORY}/scratch/
)
target_include_directories(scratch_ub-net-adapter PRIVATE
  "${ubsim_lab_root}"
  "${ubsim_lab_root}/sources/simbricks/lib")
target_compile_features(scratch_ub-net-adapter PRIVATE cxx_std_17)

build_exec(
  EXECNAME ctp-feedback-contract
  EXECNAME_PREFIX scratch_
  SOURCE_FILES ${ubsim_lab_root}/integrations/ns3ub/tests/ctp-feedback.cc
  LIBRARIES_TO_LINK ${libunified-bus} ${libcore} ${libnetwork}
  EXECUTABLE_DIRECTORY_PATH ${CMAKE_OUTPUT_DIRECTORY}/scratch/
)

build_exec(
  EXECNAME ctp-recovery-contract
  EXECNAME_PREFIX scratch_
  SOURCE_FILES ${ubsim_lab_root}/integrations/ns3ub/tests/ctp-recovery.cc
  LIBRARIES_TO_LINK ${libunified-bus} ${libcore} ${libnetwork}
  EXECUTABLE_DIRECTORY_PATH ${CMAKE_OUTPUT_DIRECTORY}/scratch/
)
